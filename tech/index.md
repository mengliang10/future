    const matchS = activeStage === 'all' || stage === activeStage;
    row.classList.toggle('tech-row-hidden', !(matchQ && matchS));
  });
  countVisible();
}

function setStage(btn, stage) {
  activeStage = stage;
  document.querySelectorAll('.stage-filter-btn').forEach(b => b.classList.remove('active'));
  btn.classList.add('active');
  filterTech();
}
